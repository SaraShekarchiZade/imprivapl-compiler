Start
    make :input: 50
    make :threshold: 100
    make :factor: 2

    Begin process(val)
        val := val * 3 - 10 / 2
    Finish

    ; Outer Loop
    repeat 3 [
        make :calc: process(input)

        ; Deep Nesting Starts
        if calc < threshold * 2 [
            print ["Inside First Block"]

            ; Math Priority in If Condition
            if 15 = 5 + 5 * 2 [
                input := input + 10

                if input > 60 [
                     make :msg: "Max Level Reached"
                     print [msg]
                ]
            ]
        ]

        input := input + factor
    ]

    final := input * (2 + 1)
    print [final]
End
